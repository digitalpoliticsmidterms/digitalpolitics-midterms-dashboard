  <script>

    const money = new Intl.NumberFormat("en-US", {

      style: "currency",

      currency: "USD",

      maximumFractionDigits: 0

    });


    const integer = new Intl.NumberFormat("en-US");


    const fullDate = value => new Date(`${value}T00:00:00`).toLocaleDateString(

      "en-US",

      { month: "short", day: "numeric", year: "numeric" }

    );


    const shortDate = value => new Date(`${value}T00:00:00`).toLocaleDateString(

      "en-US",

      { month: "short", day: "numeric" }

    );


    const escapeHtml = value => String(value).replace(/[&<>"']/g, character => ({

      "&": "&amp;",

      "<": "&lt;",

      ">": "&gt;",

      '"': "&quot;",

      "'": "&#039;"

    })[character]);


    fetch("data.json")

      .then(response => response.json())

      .then(renderDashboard);


    function renderDashboard(data) {

      const daily = data.daily_observations;

      const latest = daily[daily.length - 1];

      const previous = daily[daily.length - 2];

      const cumulative = data.cumulative;


      const change = previous

        ? ((latest.national_spend_usd - previous.national_spend_usd) /

           previous.national_spend_usd) * 100

        : null;


      document.querySelector("#metrics").innerHTML = [

        [

          "Cumulative reported spend",

          money.format(cumulative.national_spend_usd),

          `90-day baseline plus ${cumulative.daily_reports_added} later daily reports`

        ],

        [

          "Latest Daily Spend",

          money.format(latest.national_spend_usd),

          fullDate(latest.report_end_date)

        ],

        [

          "Daily % change in ad spend",

          change === null ? "—" : `${change >= 0 ? "+" : ""}${change.toFixed(1)}%`,

          previous

            ? `${shortDate(previous.report_end_date)} to ${shortDate(latest.report_end_date)}`

            : ""

        ],

        [

          "Total number of advertisers",

          integer.format(cumulative.advertiser_count),

          "Unique reported Page and disclaimer records"

        ]

      ].map(([label, value, note]) => `

        <article class="metric">

          <p class="eyebrow">${label}</p>

          <strong>${value}</strong>

          <small>${note}</small>

        </article>

      `).join("");


      document.querySelector("#daily-note").textContent =

        `${daily.length} observed reports. Missing dates are not zero-spend days.`;


      drawDailyBars(daily);

      drawStates(

        cumulative.states.slice(0, 12),

        cumulative.national_spend_usd

      );

      drawAdvertisers(latest.top_advertisers.slice(0, 10));


      document.querySelector("#methodology").textContent = data.methodology;

    }


    function drawDailyBars(rows) {

      const maximum = Math.max(

        ...rows.map(row => row.national_spend_usd)

      );


      document.querySelector("#daily-bars").innerHTML = rows.map(row => `

        <div class="daily-bar">

          <span class="value">${money.format(row.national_spend_usd)}</span>

          <div class="bar"

               style="height: ${(row.national_spend_usd / maximum) * 220}px"></div>

          <span class="date">${shortDate(row.report_end_date)}</span>

        </div>

      `).join("");

    }


    function drawStates(states, total) {

      const maximum = states[0].spend_usd;


      document.querySelector("#state-bars").innerHTML = states.map(row => `

        <div class="bar-row">

          <span>${escapeHtml(row.state)}</span>

          <div class="state-bar"

               style="width: ${(row.spend_usd / maximum) * 100}%"></div>

          <span class="amount">

            ${(row.spend_usd / total * 100).toFixed(1)}%

          </span>

        </div>

      `).join("");

    }


    function drawAdvertisers(advertisers) {

      document.querySelector("#advertisers").innerHTML = `

        <table>

          <thead>

            <tr>

              <th>Advertiser</th>

              <th class="amount">Spend</th>

              <th class="amount">Ads</th>

            </tr>

          </thead>

          <tbody>

            ${advertisers.map(row => `

              <tr>

                <td>

                  <strong>${escapeHtml(row.page_name)}</strong><br>

                  <small>${escapeHtml(row.disclaimer)}</small>

                </td>

                <td class="amount">${money.format(row.spend_usd)}</td>

                <td class="amount">${integer.format(row.ad_count)}</td>

              </tr>

            `).join("")}

          </tbody>

        </table>

      `;

    }
